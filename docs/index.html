Spis podstawowych komend i definicje

1. Podstawowe Komendy

Nazwa Argumenty Przykład Opis
print [format] [...] print("QGPU powstało w %d roku!", 2026); Funkcja podoba do funkcji printf, z dopiskiem "[QGPU] ".
qgpuCreate [width] [height] [title] [updateFunc] qgpuCreate(1280, 720, "QGPU Project", Update); Funkcja tworząca okno graficzne o danych wymiarach i tytule. (Funkcja Update jest wywoływana z każdą klatką)
drawGeometry [posX] [posY] [vertices] [vCount] [indices] [iCount] drawGeometry(0, 0, vertices, 4, indices, 6); Funkcja rysująca podaną geometrię.
drawRect [posX] [posY] [sizeX] [sizeY] [color] drawRect(0, 0, 200, 100, RED); Rysuje wypełniony prostokąt o podanych wymiarach.
drawTriangle [posX] [posY] [p1X] [p1Y] [p2X] [p2Y] [p3X] [p3Y] [color] drawTriangle(0, 0, -10, -10, 0, 10, 10, -10, GREEN); Rysuje wypełniony trójkąt o podanych wierzchołkach.
drawCircle [posX] [posY] [radius] [segments] [color] drawCircle(0, 0, 5, 16, BLUE); Rysuje koło o podanym radiusie.
drawLine [x1] [y1] [x2] [y2] [thickness] [color] drawLine(-100, 0, 100, 25, 3, DARK_RED); Rysuje linie od jednego punktu do drugiego.
drawWireRect [posX] [posY] [sizeX] [sizeY] [thickness] [color] drawWireRect(0, 0, 200, 100, 3, DARK_GREEN); Rysuje prostokąt o podanych wymiarach.
drawWireTriangle [posX] [posY] [p1X] [p1Y] [p2X] [p2Y] [p3X] [p3Y] [thickness] [color] drawWireTriangle(0, 0, -10, -10, 0, 10, 10, -10, 3, DARK_BLUE); Rysuje trójkąt o podanych wierzchołkach.
drawWireCircle [posX] [posY] [radius] [segments] [thickness] [color] drawWireCircle(0, 0, 5, 16, 3, WHITE); Rysuje okrąg o podanym radiusie.
loadTexture [filename] [slot] loadTexture(FILES "texture.qgt", 0); Ładuje podaną teksturę na podany slot.
drawTextureScale [posX] [posY] [slot] [scale] drawTextureScale(0, 0, 0, 20); Rysuje teksturę wczytaną na dany slot.
drawButton [posX] [posY] [width] [height] [color] [hoverClr] [pressClr] drawButton(0, 0, 160, 60, WHITE, GRAY, BLACK); Rysuje przycisk o podanych wymiarach.
drawSlider [posX] [posY] [width] [height] [handleW] [handleH] [value] [min] [max] [backgroundClr] [fillClr] [handleClr] drawSlider(0, 0, 160, 60, 10, 70, &sliderValue, 0, 1, GRAY, WHITE, DARK_BLUE); Rysuje suwak o podanych wymiarach.
drawToggle [posX] [posY] [width] [height] [value] [offClr] [onClr] drawToggle(0, 0, 100, 100, &toggleValue, DARK_GREEN, GREEN); Rysuje przełącznik o podanych wymiarach.
getWidth Brak getWidth(); Zwraca szerokość okna.
getHeight Brak getHeight(); Zwraca wysokość okna.
drawChar [posX] [posY] [symbol] [scale] [color] drawChar(0, 0, 'A', 20, WHITE); Rysuje pojedyńczy znak.
drawText [posX] [posY] [text] [scale] [color] drawText(0, 0, "Hello, World!", 20, WHITE); Rysuje podany tekst.
getKey [key] getKey(QKEY_W); Zwraca aktualny stan przycisku.
onKey [key] onKey(QKEY_S); Zwraca 1 w momencie kliknięcia przycisku.
getMouse [button] getMouse(LMB); Zwraca aktualny stan przycisku myszy.
onMouse [button] onMouse(RMB); Zwraca 1 w momencie kliknięcia przycisku myszy.
getMousePos [x] [y] getMousePos(&mouseX, &mouseY); Zwraca aktualną pozycję kursora.
